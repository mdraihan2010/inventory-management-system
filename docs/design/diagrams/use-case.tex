\begin{tikzpicture}[

% stick figure: \actor{id}{x,y}{label}
\newcommand{\actor}[3]{%
  \begin{scope}[shift={(#2)}]
    \draw[thick] (0,0.78) circle (0.24);
    \draw[thick] (0,0.54) -- (0,-0.08);
    \draw[thick] (-0.4,0.34) -- (0.4,0.34);
    \draw[thick] (0,-0.08) -- (-0.3,-0.6);
    \draw[thick] (0,-0.08) -- (0.3,-0.6);
    \node[font=\small\bfseries] at (0,-0.92) {#3};
    \coordinate (#1) at (0,0.2);
    \coordinate (#1-top) at (0,1.05);
    \coordinate (#1-bottom) at (0,-1.2);
  \end{scope}}

% ---------------- left column: Manager (and Administrator) ----------------
\def\L{5.6}
\node[usecase] (products)   at (\L,13.20) {Manage products\\(add, edit, deactivate)};

\def\R{11.6}
\node[usecase] (users)      at (\R,13.20) {Manage users\\(create, role, deactivate)};

\node[shared] (stock)       at (7.2,-1.45) {Update stock and record\\stock movement};

% ---------------- actors ----------------

\actor{admin}{17.6,12.05}{Administrator}

% ---------------- associations ----------------
\begin{scope}[on background layer/.style={}]
\foreach \u in {products,catbrand,suppliers,editcust,reports,dues,returns,cancelret,canceldraft,paysup,adjust,purchase}
  { \draw[assoc] ($(manager)+(0.45,0)$) -- (\u.west); }
\foreach \u in {users,resetpw,audit}
  { \draw[assoc] ($(admin)+(-0.45,0)$) -- (\u.east); }
\foreach \u in {login,password,view,addcust,sale,collect,invoice,complete}
  { \draw[assoc] ($(staff)+(-0.45,0)$) -- (\u.east); }
\end{scope}

% ---------------- generalization: Administrator is a Manager, Manager is a Staff ----------------

\node[font=\scriptsize\itshape, fill=white, inner sep=1pt] at (8.5,-3.33) {Manager can do everything a Staff user can};
